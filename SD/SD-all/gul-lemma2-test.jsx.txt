import { useState, useEffect, useRef, useCallback } from "react";

// ── Math ─────────────────────────────────────────────────────────────────────
const clamp = (v, lo, hi) => Math.max(lo, Math.min(hi, v));
const norm2 = (ax, ay) => Math.sqrt(ax * ax + ay * ay);

// Surface entropy: Gaussian centred at (xc, yc)
const SB = (x, y, xc = 0, yc = 0) => Math.exp(-((x - xc) ** 2 + (y - yc) ** 2));
const gradSB = (x, y, xc = 0, yc = 0) => {
  const s = SB(x, y, xc, yc);
  return [-2 * (x - xc) * s, -2 * (y - yc) * s];
};

// Curvature terms
const curvatureModes = {
  zero:       { label: "R = 0 (baseline)",           fn: () => [0, 0] },
  mild:       { label: "R = sin(x)cos(y)  [mild]",   fn: (x, y) => [Math.cos(x) * Math.cos(y), -Math.sin(x) * Math.sin(y)] },
  violating:  { label: "R = −3·∇S_B  [VIOLATES L2]", fn: (x, y, xc, yc) => { const [gx, gy] = gradSB(x, y, xc, yc); return [-3 * gx, -3 * gy]; } },
  adversarial:{ label: "R = steep anti-aligned  [VIOLATES L2]", fn: (x, y, xc, yc) => {
    // Steep slope directly opposing grad S_B near origin
    const [gx, gy] = gradSB(x, y, xc, yc);
    const mag = norm2(gx, gy) || 1e-9;
    return [-4 * gx / mag, -4 * gy / mag];  // unit anti-alignment, scaled ×4
  }},
};

// Successor vector: S_vec = T·gradSB - beta·gradR
const svec = (x, y, T, beta, curvMode, xc = 0, yc = 0) => {
  const [sgx, sgy] = gradSB(x, y, xc, yc);
  const [rgx, rgy] = curvatureModes[curvMode].fn(x, y, xc, yc);
  return [T * sgx - beta * rgx, T * sgy - beta * rgy];
};

// Lemma 2 bound check: returns {satisfied, ratio, delta, epsilon}
const lemma2Check = (x, y, T, beta, curvMode, xc = 0, yc = 0) => {
  const [sgx, sgy] = gradSB(x, y, xc, yc);
  const [rgx, rgy] = curvatureModes[curvMode].fn(x, y, xc, yc);
  const normGradSB = norm2(sgx, sgy);
  const normBetaR  = beta * norm2(rgx, rgy);
  if (normGradSB < 1e-8) return { satisfied: true, ratio: 0, delta: 1, epsilon: T * T };
  const ratio = normBetaR / (T * normGradSB);
  const satisfied = ratio <= 1;
  const delta = Math.max(0, 1 - ratio);
  const epsilon = (delta * T) ** 2;
  return { satisfied, ratio, delta, epsilon };
};

// RK4 integrator step
const rk4Step = (x, y, dt, T, beta, curvMode, xc, yc) => {
  const f = (px, py) => svec(px, py, T, beta, curvMode, xc, yc);
  const [k1x, k1y] = f(x, y);
  const [k2x, k2y] = f(x + 0.5 * dt * k1x, y + 0.5 * dt * k1y);
  const [k3x, k3y] = f(x + 0.5 * dt * k2x, y + 0.5 * dt * k2y);
  const [k4x, k4y] = f(x + dt * k3x, y + dt * k3y);
  return [
    x + (dt / 6) * (k1x + 2 * k2x + 2 * k3x + k4x),
    y + (dt / 6) * (k1y + 2 * k2y + 2 * k3y + k4y),
  ];
};

// Integrate a trajectory from (x0,y0) for nSteps
const integrate = (x0, y0, T, beta, curvMode, xc, yc, nSteps = 400, dt = 0.04) => {
  const pts = [[x0, y0]];
  let x = x0, y = y0;
  for (let i = 0; i < nSteps; i++) {
    [x, y] = rk4Step(x, y, dt, T, beta, curvMode, xc, yc);
    x = clamp(x, -5, 5); y = clamp(y, -5, 5);
    pts.push([x, y]);
    if (norm2(x - xc, y - yc) < 0.05) break;  // converged
  }
  return pts;
};

// Starting points: ring around domain
const STARTS = Array.from({ length: 16 }, (_, i) => {
  const a = (i / 16) * 2 * Math.PI;
  return [3.5 * Math.cos(a), 3.5 * Math.sin(a)];
}).concat([[-2, 2], [2, -2], [-2, -2], [2, 2], [0, 3], [3, 0], [-3, 0], [0, -3]]);

// ── Canvas renderer ───────────────────────────────────────────────────────────
const W = 480, H = 480;
const toCanvas = (x, y) => [(x + 5) / 10 * W, H - (y + 5) / 10 * H];
const fromCanvas = (px, py) => [px / W * 10 - 5, (H - py) / H * 10 - 5];

function drawScene(canvas, T, beta, curvMode, xc, yc, showBound, showField) {
  const ctx = canvas.getContext("2d");
  ctx.clearRect(0, 0, W, H);

  // Background: S_B heatmap
  const img = ctx.createImageData(W, H);
  for (let py = 0; py < H; py++) {
    for (let px = 0; px < W; px++) {
      const [x, y] = fromCanvas(px, py);
      const sb = SB(x, y, xc, yc);
      const { satisfied } = showBound
        ? lemma2Check(x, y, T, beta, curvMode, xc, yc)
        : { satisfied: true };
      const i = (py * W + px) * 4;
      if (!satisfied && showBound) {
        // Violated region: red tint
        img.data[i]     = Math.round(180 + sb * 60);
        img.data[i + 1] = Math.round(20);
        img.data[i + 2] = Math.round(20);
        img.data[i + 3] = 255;
      } else {
        // Normal: navy-blue entropy heatmap
        img.data[i]     = Math.round(sb * 20);
        img.data[i + 1] = Math.round(sb * 60);
        img.data[i + 2] = Math.round(sb * 180 + (1 - sb) * 8);
        img.data[i + 3] = 255;
      }
    }
  }
  ctx.putImageData(img, 0, 0);

  // Grid
  ctx.strokeStyle = "rgba(255,255,255,0.06)";
  ctx.lineWidth = 0.5;
  for (let i = 0; i <= 10; i++) {
    const px = i / 10 * W, py = i / 10 * H;
    ctx.beginPath(); ctx.moveTo(px, 0); ctx.lineTo(px, H); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0, py); ctx.lineTo(W, py); ctx.stroke();
  }

  // Vector field (sparse)
  if (showField) {
    const step = 20;
    for (let px = step / 2; px < W; px += step) {
      for (let py = step / 2; py < H; py += step) {
        const [x, y] = fromCanvas(px, py);
        const [vx, vy] = svec(x, y, T, beta, curvMode, xc, yc);
        const mag = norm2(vx, vy);
        if (mag < 1e-6) continue;
        const len = clamp(mag * 6, 2, 11);
        const nx = vx / mag, ny = vy / mag;
        const ex = px + nx * len, ey = py - ny * len;
        const hue = (1 - clamp(mag * 4, 0, 1)) * 200;
        ctx.strokeStyle = `hsla(${hue},80%,65%,0.55)`;
        ctx.lineWidth = 0.8;
        ctx.beginPath(); ctx.moveTo(px, py); ctx.lineTo(ex, ey); ctx.stroke();
        // Arrowhead
        const ang = Math.atan2(-ny, nx);
        ctx.fillStyle = `hsla(${hue},80%,65%,0.55)`;
        ctx.beginPath();
        ctx.moveTo(ex, ey);
        ctx.lineTo(ex - 3.5 * Math.cos(ang - 0.4), ey - 3.5 * (-Math.sin(ang - 0.4)));
        ctx.lineTo(ex - 3.5 * Math.cos(ang + 0.4), ey - 3.5 * (-Math.sin(ang + 0.4)));
        ctx.closePath(); ctx.fill();
      }
    }
  }

  // Trajectories
  const results = [];
  for (const [x0, y0] of STARTS) {
    const pts = integrate(x0, y0, T, beta, curvMode, xc, yc);
    const last = pts[pts.length - 1];
    const dist = norm2(last[0] - xc, last[1] - yc);
    const converged = dist < 0.3;
    results.push({ pts, converged, start: [x0, y0], end: last, dist });

    const color = converged ? "rgba(74,222,128,0.85)" : "rgba(248,113,113,0.85)";
    ctx.strokeStyle = color;
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    for (let i = 0; i < pts.length; i++) {
      const [cx, cy] = toCanvas(pts[i][0], pts[i][1]);
      i === 0 ? ctx.moveTo(cx, cy) : ctx.lineTo(cx, cy);
    }
    ctx.stroke();

    // Start dot
    const [spx, spy] = toCanvas(x0, y0);
    ctx.fillStyle = color;
    ctx.beginPath(); ctx.arc(spx, spy, 3, 0, Math.PI * 2); ctx.fill();

    // End dot
    const [epx, epy] = toCanvas(last[0], last[1]);
    ctx.fillStyle = converged ? "#4ade80" : "#f87171";
    ctx.strokeStyle = "#fff";
    ctx.lineWidth = 1;
    ctx.beginPath(); ctx.arc(epx, epy, 4, 0, Math.PI * 2);
    ctx.fill(); ctx.stroke();
  }

  // Entropy peak marker
  const [pcx, pcy] = toCanvas(xc, yc);
  ctx.strokeStyle = "#fff";
  ctx.lineWidth = 2;
  ctx.beginPath(); ctx.arc(pcx, pcy, 10, 0, Math.PI * 2); ctx.stroke();
  ctx.fillStyle = "rgba(255,255,255,0.9)";
  ctx.font = "bold 11px monospace";
  ctx.fillText("x*", pcx + 13, pcy + 4);

  return results;
}

// ── App ───────────────────────────────────────────────────────────────────────

    if (!canvas) return;

  const noneConverge = convergedCount === 0;

    : { text: `⚠ PARTIAL — ${convergedCount}/${totalCount} converge`, color: "#fbbf24", bg: "rgba(120,53,15,0.5)" };

  const sl = { display: "flex", flexDirection: "column", gap: 3 };
  const labelStyle = { color: "#94a3b8", fontSize: 11, fontFamily: "monospace" };
  const valStyle = { color: "#7dd3fc", fontFamily: "monospace", fontSize: 13, fontWeight: 700 };

  return (
    <div style={{
      minHeight: "100vh",
      background: "linear-gradient(160deg, #020810 0%, #050d1a 60%, #020810 100%)",
      color: "#e2e8f0",
      fontFamily: "'JetBrains Mono', 'Courier New', monospace",
      padding: "20px",
      boxSizing: "border-box",
    }}>
      {/* Header */}
      <div style={{ marginBottom: 16 }}>
        <div style={{ display: "flex", alignItems: "baseline", gap: 12, flexWrap: "wrap" }}>
          <h1 style={{ margin: 0, fontSize: 17, fontWeight: 700, color: "#f1f5f9", letterSpacing: "0.06em" }}>
            GUL LEMMA 2 — NUMERICAL FALSIFICATION TEST
          </h1>
          <span style={{ color: "#334155", fontSize: 11 }}>RK4 · 24 trajectories · live</span>
        </div>
        <p style={{ margin: "5px 0 0", fontSize: 11, color: "#475569" }}>
          Hypothesis: <code style={{ color: "#7dd3fc" }}>‖β∇R‖ ≤ (1−δ)T‖∇S_B‖</code> outside U
          &nbsp;→&nbsp;
          all trajectories converge to entropy peak x*
        </p>
      </div>

      <div style={{ display: "flex", gap: 16, flexWrap: "wrap", alignItems: "flex-start" }}>
        {/* Canvas */}
        <div style={{ position: "relative" }}>
          <canvas ref={canvasRef} width={W} height={H}
            style={{ borderRadius: 6, border: "1px solid rgba(255,255,255,0.08)", display: "block" }} />
          {/* Overlay legend */}
          <div style={{
            position: "absolute", bottom: 10, left: 10,
            background: "rgba(0,0,0,0.65)", padding: "6px 10px",
            borderRadius: 4, fontSize: 10, fontFamily: "monospace", lineHeight: 1.8,
          }}>
            <div><span style={{ color: "#4ade80" }}>●</span> converged to x*</div>
            <div><span style={{ color: "#f87171" }}>●</span> failed to converge</div>
            {showBound && <div><span style={{ color: "rgba(255,80,80,0.7)" }}>■</span> L2 bound violated</div>}
          </div>
        </div>

        {/* Controls + Results */}
        <div style={{ flex: 1, minWidth: 260, display: "flex", flexDirection: "column", gap: 14 }}>

          {/* Verdict */}

            background: verdict.bg,

          }}>

            <div style={{ color: "#94a3b8", fontSize: 11, marginTop: 4 }}>
              {convergedCount}/{totalCount} trajectories reached ‖x − x*‖ &lt; 0.3
            </div>
          </div>

          {/* Curvature mode */}
          <div>
            <div style={{ ...labelStyle, marginBottom: 6 }}>CURVATURE MODE  R(x,y)</div>
            {Object.entries(curvatureModes).map(([key, { label }]) => (
              <label key={key} style={{
                display: "flex", alignItems: "center", gap: 8, cursor: "pointer",
                padding: "5px 8px", borderRadius: 4, marginBottom: 3,

          {/* Toggles */}

            {[["Show bound violation", showBound, setShowBound], ["Show vector field", showField, setShowField]].map(([label, val, set]) => (

                {label}

          </div>

          {/* Lemma 2 numerical check at sample point */}
          {sampleCheck && (

          {/* Interpretation */}
